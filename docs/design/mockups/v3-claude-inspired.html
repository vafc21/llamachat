// Rust file watcher — built with LlamaChat
use notify::{Watcher, RecursiveMode, watcher};
use notify_debouncer_mini::{new_debouncer};
use std::process::Command;
use std::path::PathBuf;
use clap::Parser;
use serde::Deserialize;
#[derive(Parser)]
struct Args {
#[arg(short, long, default_value = "watch.yaml")]
config: PathBuf,
}
#[derive(Deserialize, Clone)]
struct WatchTarget {
pattern: String,
command: String,
#[serde(default = "default_debounce")]
debounce_ms: u64,
}
fn default_debounce() -> u64 { 500 }
fn main() -> anyhow::Result<()> {
let args = Args::parse();
let config: Vec<WatchTarget> = serde_yaml::from_str(
&std::fs::read_to_string(&args.config)?
)?;
let (tx, rx) = std::sync::mpsc::channel();
let mut debouncer = new_debouncer(
Duration::from_millis(200), tx
)?;
// Watch each target directory
for target in &config {
debouncer.watcher()
.watch(&PathBuf::from(&target.pattern), RecursiveMode::Recursive)?;
}
println!("👀 Watching for changes…");
// Event loop
for result in rx {
match result {
Ok(_events) => {
for target in &config {
Command::new("sh").arg("-c")
.arg(&target.command).status()?;
}
}
Err(e) => eprintln!("Watch error: {e:?}"),
}
}
}
